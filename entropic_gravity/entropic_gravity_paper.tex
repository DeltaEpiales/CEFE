\section{Introduction}

The entropic-gravity hypothesis holds that spacetime geometry and gravity are
not fundamental but emerge from the thermodynamics of information associated
with holographic screens
\cite{Bekenstein1973,Hawking1975,Jacobson1995,Verlinde2011,Padmanabhan2010}.
Its modern form combines Bekenstein's area law for black-hole entropy,
Unruh's observer-dependent temperature \cite{Unruh1976}, Jacobson's derivation
of the Einstein equation from local horizon thermodynamics \cite{Jacobson1995},
and Verlinde's explicit construction of Newtonian gravity as an entropic force
\cite{Verlinde2011}. A separate line of work established that the entanglement
entropy of quantum fields obeys an area law quite generally
\cite{Bombelli1986,Srednicki1993,Plenio2005,CasiniHuerta2009}, providing the
microscopic candidate for the ``bits'' on the screen.

This paper does two things. First, it distills the hypothesis into a minimal
set of axioms and derives the testable consequences as propositions with
proofs. Second, it measures every consequence that a lattice field-theory
simulator can measure, using the CEFE engine \cite{CEFE}: a C++ implementation
of the exact Srednicki covariance-matrix method on causal-diamond lattices,
validated against the published literature \cite{Srednicki1993,Bombelli1986}
and against the lattice methods of \cite{Gupta1998,Lepage2005}. The lattice
spacing $a$ plays the role of the fundamental (Planck) length throughout.

\section{Axioms and Framework}

\begin{definition}[Holographic screen]
A \emph{screen} is the boundary $\partial A$ of a spatial subregion $A$ of a
constant-time slice of a causal diamond. The screen of radius $R$ encloses the
mass-energy $M$ that has emerged in the interior.
\end{definition}

\begin{axiom}[Area scaling of information]
\label{ax:area}
The information content of a screen scales with its area in units of the
fundamental length, $S = c_2\, A/a^2$, with $c_2$ an $O(1)$ constant fixed by
the field content.
\end{axiom}

\begin{axiom}[Equipartition and the entropy gradient]
\label{ax:verlinde}
The screen's bits $N = 4S$ share the enclosed energy equally,
$\tfrac12 N k_B T = Mc^2$; a test mass $m$ displaced by $\Delta x$ near the
screen changes the entropy linearly, $\Delta S = 2\pi k_B (mc/\hbar)\,\Delta x$.
\end{axiom}

Axiom~\ref{ax:area} is \emph{testable} --- it is the subject of
Proposition~\ref{prop:area}. The bit identification $N = 4S$ and
Axiom~\ref{ax:verlinde} are \emph{not testable} on any fixed-background
simulator; they are the irreducible postulates of the theory
(Section~\ref{sec:axioms}). What \emph{is} testable is whether, given the
measured $S(R)$, the chain of Axiom~\ref{ax:verlinde} yields Newtonian gravity
(Proposition~\ref{prop:force}).

\begin{definition}[Covariance-matrix entropy \cite{Srednicki1993,Bombelli1986}]
On a lattice with spacing $a$, let $W = -\nabla^2 + m^2$ be the positive
operator of a free scalar Hamiltonian
$H = \tfrac12\pi^2 + \tfrac12\phi\, W\phi$. The vacuum covariance matrices are
$C = \tfrac12 W^{-1/2}$ and $P = \tfrac12 W^{+1/2}$; at temperature $T$ both
acquire the factor $\coth(\omega/2T)$. For a subregion $A$, the entanglement
entropy is
\begin{equation}
S_A = \sum_k \Big[\big(\nu_k+\tfrac12\big)\ln\big(\nu_k+\tfrac12\big)
-\big(\nu_k-\tfrac12\big)\ln\big(\nu_k-\tfrac12\big)\Big],
\label{eq:srednicki}
\end{equation}
where $\nu_k^2$ are the eigenvalues of $C_A \cdot P_A$ restricted to $A$.
\end{definition}

\section{Proposition 2: Emergence of the Inverse-Square Force}
\label{prop:force}

\begin{proposition}[Verlinde loop \cite{Verlinde2011}]
Given Axiom~\ref{ax:area} with measured constant $c_2$ and
Axiom~\ref{ax:verlinde}, the entropic force between the enclosed mass $M$ and
a test mass $m$ at screen radius $R$ is
\begin{equation}
F = \frac{G_{\mathrm{eff}}\, M m}{R^2},
\qquad
G_{\mathrm{eff}} = \frac{a^2}{4 c_2},
\end{equation}
and in the Bekenstein normalization $c_2 = 1/4$, $G = a^2 = \ell_{\mathrm P}^2$
exactly.
\end{proposition}

\begin{proof}
The bit count on the screen is ($k_B = 1$)
\[
N = 4S = 4 c_2 \frac{A}{a^2} = \frac{16\pi c_2 R^2}{a^2}.
\]
Equipartition of the enclosed energy, $\tfrac12 N T = M$ (natural units
$c = \hbar = k_B = 1$), gives the screen temperature
\[
T(R) = \frac{2M}{N} = \frac{M a^2}{8\pi c_2 R^2}.
\]
The entropy gradient of Axiom~\ref{ax:verlinde} is
$\dd S/\dd x = 2\pi m$, constant in $R$. The entropic force
$F\,\Delta x = T\,\Delta S$ therefore gives
\[
F(R) = T\,\frac{\dd S}{\dd x}
= \frac{M a^2}{8\pi c_2 R^2}\cdot 2\pi m
= \frac{a^2}{4 c_2}\,\frac{M m}{R^2},
\]
which is Newton's law with $G_{\mathrm{eff}} = a^2/(4c_2)$. Setting
$c_2 = 1/4$ (the Bekenstein--Hawking normalization $S = A/4a^2$) recovers
$G = a^2$, the identification of the fundamental length with the Planck
length.
\end{proof}

\begin{corollary}[Entanglement renormalization of $G$ \cite{SusskindUglum1994}]
For a single real scalar, the measured $c_2 = 0.0250$ implies
$G_{\mathrm{eff}} = 10.0\,a^2$: one scalar species carries
$\approx 10\%$ of the Bekenstein bit density. This is precisely the
Susskind--Uglum mechanism, in which the entanglement divergence renormalizes
$1/G$; $\sim 10$ field species (or the Standard Model content) saturate the
Bekenstein normalization.
\end{corollary}

I(A\!:\!B) = S_A + S_B - S_{A\cup B}

\end{proposition}

\begin{proof}[Proof sketch]
The leading divergence of a free-field entanglement entropy is local on the

divergences cancel pairwise in $I(A\!:\!B)$; subleading smooth-boundary
corrections cancel likewise, and the remainder is suppressed by the mass gap,

\end{proof}

\section{Proposition 4: The Entanglement First Law}
\label{prop:firstlaw}

\begin{proposition}[$\delta S = \delta\langle K\rangle$]
\label{thm:firstlaw}
Let $\rho(\lambda)$ be a differentiable family of states with
$\rho(0) = \rho_0$, and let $K_0 = -\ln\rho_0$ be the modular Hamiltonian of
$\rho_0$. Then to first order in $\lambda$,
\begin{equation}
\delta S = \delta\langle K_0\rangle + O(\lambda^2).
\end{equation}
\end{proposition}

\begin{proof}
The relative entropy is
\[
S(\rho_1\|\rho_0)
= \operatorname{tr}\rho_1\ln\rho_1 - \operatorname{tr}\rho_1\ln\rho_0
= -S(\rho_1) + \langle K_0\rangle_{\rho_1}.
\]
Evaluating at $\rho_1 = \rho_0$ gives
$0 = -S(\rho_0) + \langle K_0\rangle_{\rho_0}$. Subtracting,
\[
S(\rho_1\|\rho_0)
= \big(\langle K_0\rangle_{\rho_1} - \langle K_0\rangle_{\rho_0}\big)
- \big(S(\rho_1) - S(\rho_0)\big)
= \delta\langle K_0\rangle - \delta S.
\]
Relative entropy is nonnegative and vanishes at $\lambda = 0$; being a
smooth function of $\lambda$ minimized at $\lambda = 0$, its first derivative
there is zero, so $S(\rho_1\|\rho_0) = O(\lambda^2)$ and
$\delta S = \delta\langle K_0\rangle + O(\lambda^2)$.
\end{proof}

\noindent
This identity is the equilibrium condition at the heart of Jacobson's
argument that the Einstein equation is an equation of state
\cite{Jacobson1995,Jacobson2016}.

\end{definition}

\section{Proposition 5: The Arrow of Time is Consistent with Unitarity}
\label{prop:arrow}

\begin{proposition}
Under unitary evolution, fine-grained (von Neumann) entropy is exactly
conserved, while coarse-grained entropy can grow. The entropic-time axiom
(time as accumulated entropy \cite{OGHoTT,Barontini2025}) is therefore
consistent with quantum mechanics.
\end{proposition}

\begin{proof}
$\rho(t) = U(t)\rho(0)U^\dagger(t)$ has the same spectrum as $\rho(0)$, so
every spectral function, including $S = -\operatorname{tr}\rho\ln\rho$ and the
purity $\operatorname{tr}\rho^2$, is exactly conserved. Coarse-grained
entropies are not spectral functions and are unconstrained.
\end{proof}

\end{observation}
